# Azure pipeline -- every package it needs, and why.
# Install with:  pip install -r requirements.txt

# --- Azure services ---
openai                          # includes AzureOpenAI, the client for Azure OpenAI
azure-search-documents          # Azure AI Search: create the index, upload chunks, search
azure-ai-documentintelligence   # Azure AI Document Intelligence: PDF -> markdown
azure-storage-blob              # Azure Blob Storage: upload and download the PDFs

# --- pipeline ---
langgraph                       # the graph: steps, routing, and the pause for human review
langchain-text-splitters        # cuts markdown into chunks at headings and paragraphs
numpy                           # maths for the topic check's similarity scores
python-dotenv                   # reads keys and endpoints from the .env file

# --- tests ---
pytest                          # runs the offline tests in tests/
